on_actions = {
    # 每日更新能识别吞并、解放、迁都和边境控制变化；旧档已完成国策也会自动恢复入口。
    on_daily = {
        effect = {
            # on_daily 对各国分别运行；仅允许 FAT 自己刷新一次，避免每个国家都触发遍历。
            if = {
                limit = { tag = FAT FAT_unification_unlocked = yes }
                set_country_flag = FAT_military_liberation_unlocked
                FAT_refresh_military_liberation_targets = yes
            }
        }
    }
}
