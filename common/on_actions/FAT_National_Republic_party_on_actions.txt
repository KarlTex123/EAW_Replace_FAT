on_actions = {
	on_startup = {
		effect = {
			FAT = {
				add_dynamic_modifier = {
					modifier = FAT_National_Republic_party
				}
				set_country_flag = FAT_National_Republic_party_democratic_in_union
				set_country_flag = FAT_National_Republic_party_neutrality_in_union
				set_country_flag = FAT_National_Republic_party_communism_in_union
				if = { #谐律派系：25%稳定度
					limit = {
						has_country_flag = FAT_National_Republic_party_democratic_in_union
					}
					set_variable = { FAT_National_Republic_party_union_democratic = ROOT.party_popularity@democratic }
					clamp_variable = {
						var = FAT_National_Republic_party_union_democratic
						min = 0
						max = 0.25
					}
				}
				if = {
					limit = {
						NOT = {
							has_country_flag = FAT_National_Republic_party_democratic_in_union
						}
						NOT = {
							check_variable = { FAT_National_Republic_party_union_democratic = 0 }
						}
					}
					set_variable = { FAT_National_Republic_party_union_democratic = 0 }
				}
				if = { #中立派系：25%工厂产出
					limit = {
						has_country_flag = FAT_National_Republic_party_neutrality_in_union
					}
					set_variable = { FAT_National_Republic_party_union_neutrality = ROOT.party_popularity@neutrality }

						var = FAT_National_Republic_party_union_neutrality
						min = 0
						max = 0.25
					}
				}
				if = { 
					limit = {
						NOT = {
							has_country_flag = FAT_National_Republic_party_neutrality_in_union
						}
						NOT = {
							check_variable = { FAT_National_Republic_party_union_neutrality = 0 }
						}
					}
					set_variable = { FAT_National_Republic_party_union_neutrality = 0 }
				}
				if = { #共产派系：1点pp获取
					limit = {
						has_country_flag = FAT_National_Republic_party_communism_in_union
					}
					set_variable = { FAT_National_Republic_party_union_communism = ROOT.party_popularity@communism }
					multiply_variable = { FAT_National_Republic_party_union_communism = 4 }
					clamp_variable = {
						var = FAT_National_Republic_party_union_communism
						min = 0
						max = 1
					}
				}
				if = {
					limit = {
						NOT = {
							has_country_flag = FAT_National_Republic_party_communism_in_union
						}
						NOT = {
							check_variable = { FAT_National_Republic_party_union_communism = 0 }
						}
					}
					set_variable = { FAT_National_Republic_party_union_communism = 0 }
				}
			}
		}
	}
