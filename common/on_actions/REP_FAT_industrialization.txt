on_actions = {
	on_weekly = {
		effect = {
			if = {
				limit = {
					num_of_factories > 50
					tag = FAT
				}
				swap_ideas = {
					remove_idea = the_first_step_of_industrialization
					add_idea = Moderately_developed_industrialization
				}
				country_event = {id = FAT_start.9}
			}
			if = {
				limit = {
					num_of_factories > 80
					tag = FAT
				}
				swap_ideas = {
					remove_idea = Moderately_developed_industrialization
					add_idea = Booming_industrialization
				}
				country_event = {id = FAT_start.9}
			}
			if = {
				limit = {
					num_of_factories > 150
					tag = FAT
				}
				swap_ideas = {
					remove_idea = Booming_industrialization
					add_idea = Industrial_heart
				}
				country_event = {id = FAT_start.9}
			}
		}
	}
}
