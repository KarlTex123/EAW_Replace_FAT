#检查派系所支持的意识形态是否在执政联盟 > 设置倍率
FAT_check_set_buff_ideo_factor = {
	for_each_scope_loop = {
		array = FAT.FAT_faction_array
		set_variable = {
			if_in_union = 0
		}
		set_variable = {
			buff_ideo_factor = 0
		}
		if = {
			limit = {
				is_in_array = {
					array = FAT.FAT_government_union
					value = support_ideology
				}
			}
			set_variable = {
				if_in_union = 1
			}
			if = {
				limit = {
					check_variable = {
						support_ideology = 1
					}
				}
				set_variable = {
					buff_ideo_factor = FAT.party_popularity@democratic
				}
			}
			if = {
				limit = {
					check_variable = {
						support_ideology = 2
					}
				}
				set_variable = {
					buff_ideo_factor = FAT.party_popularity@communism
				}
			}
			if = {
				limit = {
					check_variable = {
						support_ideology = 3
					}
				}
				set_variable = {
					buff_ideo_factor = FAT.party_popularity@fascism
				}
			}
			if = {
				limit = {
					check_variable = {
						support_ideology = 4
					}
				}
				set_variable = {
					buff_ideo_factor = FAT.party_popularity@neutrality
				}
			}
		}
	}
}
FAT_set_faction_buff = {
	for_each_scope_loop = {
		array = FAT_faction_array
		set_variable = {
			buff = buff_base
		}
		multiply_variable = {
			buff = buff_ideo_factor
		}
		multiply_variable = {
			buff = buff_factor
		}
	}
	for_each_loop = {
		array = FAT_faction_array
		value = v
		index = i
		var:v = {
			set_variable = {
				FAT.FAT_faction_buff_array^i = buff
			}
		}
	}
}
