    icon = x
    cost = 50
    days_remove = 20
    allowed = {
        original_tag = FAT
    }
    fire_only_once = no
    available = {
		has_country_flag = AFITR_route_astler
        NOT = {
            has_country_flag = AFITR_mechanism_over
        }
	}
    visible = {
        has_country_flag = AFITR_route_astler
        NOT = {
            has_country_flag = AFITR_mechanism_over
        }
    }
    remove_effect = {
        add_to_variable = {
            AFITR_suntail_power = -5
        }
        add_to_variable = {
            AFITR_sunglider_power = -5
        }
        clamp_variable = {
            var = AFITR_suntail_power
            min = 0
            max = 100
        }
        clamp_variable = {
            var = AFITR_sunglider_power
            min = 0
            max = 100
        }
        add_to_variable = {
            AFITR_suppression_count = 1 #累计压制次数，3次触发军队干涉
        }
    }
}

#选举倒计时（180天，完成时触发最高清算或共和国的死局，机制结束）
AFITR_election_countdown = {
    icon = x
    days_mission_timeout = 180
    available = {
		always = no
	}
    activation = {
        has_country_flag = FAT_a_fork_in_the_road
		NOT = {
			has_country_flag = AFITR_mechanism_over
		}
	}
        timeout_effect = {
            custom_effect_tooltip = AFITR_election_countdown_tt
            hidden_effect = {
            IF = {
                limit = {
                    check_variable = { AFITR_suppression_count > 2 } #只要完成过三次压制，无论谁赢，军队都会干涉
                }
                country_event = { id = FAT_AFITR.5 } #共和国的死局：军队发动军事政变
            }
            ELSE = {
                country_event = { id = FAT_AFITR.4 } #最高清算：势力最高的派系夺下共和国的遗产
            }
        }
    }
}
