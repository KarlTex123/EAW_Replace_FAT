# 特殊国家同时保留普通扩张和友好外交两种决议，不互相排除。
REP_FAT_unification_category = {
    FAT_military_liberate_country = { # 处理[FROM.GetName]问题；保留已发布的决议 ID。
        icon = generic_prepare_civil_war
        target_array = FAT_military_liberation_targets
        target_root_trigger = { FAT_unification_unlocked = yes }
        target_trigger = { FROM = { FAT_unification_target = yes } }
        visible = { FAT_unification_unlocked = yes }
        available = {
            FROM = { NOT = { has_country_flag = FAT_unification_pending } }
            custom_trigger_tooltip = {
                tooltip = FAT_unification_can_process_tt
                FROM = {
                    OR = {
                        has_government = democratic
                        FAT_unification_war_possible = yes
                    }
                }
            }
        }
        cost = 50
        fire_only_once = no
        complete_effect = {
            # 本地事件目标随各自事件链保存，多国事件同时打开时不互相覆盖。
            FROM = {
                save_event_target_as = FAT_unification_event_target
                set_country_flag = FAT_unification_pending
            }
            if = {
                limit = { FROM = { has_government = democratic } }
                country_event = { id = FAT_Unification.1 }
            }
            else = { country_event = { id = FAT_Unification.2 } }
        }
        ai_will_do = { factor = 10 }
    }
    FAT_unification_offer_friendship = { # 向[FROM.GetName]伸出友谊的橄榄枝
        icon = generic_political_discourse
        target_array = FAT_unification_friendship_targets
        target_root_trigger = { FAT_unification_unlocked = yes }
        target_trigger = { FROM = { FAT_unification_friendship_target = yes } }
        visible = { FAT_unification_unlocked = yes }
        available = { FROM = { NOT = { has_country_flag = FAT_unification_friendship_completed } } }
        cost = 0
        fire_only_once = yes
        complete_effect = {
            add_opinion_modifier = { target = FROM modifier = FAT_unification_friendship_opinion }
            FROM = {
                add_opinion_modifier = { target = ROOT modifier = FAT_unification_friendship_opinion }
                set_country_flag = FAT_unification_friendship_completed
            }
            diplomatic_relation = { country = FROM relation = non_aggression_pact active = yes }
        }
        ai_will_do = { factor = 10 }
    }
}
