country_event = { #最高清算：势力最高的派系将夺下共和国的遗产
    id = FAT_AFITR.4
    title = FAT_AFITR.4.t
    desc = FAT_AFITR.4.d
    picture = GFX_REP_FAT_event
    is_triggered_only = yes
    immediate = {
        set_country_flag = AFITR_mechanism_over #机制结束
        clr_country_flag = FAT_a_fork_in_the_road #收起命运分歧决议组
        hidden_effect = { #哪个派系赢了就切换对应的领导人
            if = {
                limit = {
                    check_variable = { AFITR_astler_power > AFITR_suntail_power }
                    check_variable = { AFITR_astler_power > AFITR_sunglider_power }
                }
                set_politics = { ruling_party = communism } #鹫民阵线（民主社会主义）属共产主义组
                promote_character = FAT_rikard_astler #艾斯特勒
            }
            else_if = {
                limit = {
                    check_variable = { AFITR_suntail_power > AFITR_sunglider_power }
                }
                set_politics = { ruling_party = fascism } #民族阵线（极端民族主义）属霸权主义组
                promote_character = FAT_reinhard_suntail #曜尾
            }
            else = {
                set_politics = { ruling_party = democratic } #谐律同盟（自由主义）属谐律组
                promote_character = FAT_schnabel_sunglider #翔日
            }
        }
    }
    option = {
        name = FAT_AFITR.4.1
        custom_effect_tooltip = AFITR_victory_wip_tt #悬停提示：内容未完成
        load_focus_tree = generic_griffon_focus #移除整个国策树，切换为狮鹫尼亚通用国策树
    }
}
